let mode=null, activeCat='signature', cart={};

 

function showPage(id){

    document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));

    document.getElementById(id).classList.add('active');

    window.scrollTo(0,0);

}

function goCover(){ showPage('cover'); document.getElementById('cartBar').classList.add('hidden'); }

function enterMode(m){

    mode=m;

    document.getElementById('modeTag').textContent = m==='dinein' ? '大厅堂食 · 自助点单' : '客房点餐 · 送房上桌';

    activeCat='signature';

    renderCatBar(); renderDishes();

    showPage('orderPage');

}

function showRooms(){ showPage('roomPage'); renderRooms(); }

 

function renderCatBar(){

    document.getElementById('catBar').innerHTML = CATS.map(c=>`<div class="cat-tab ${c.id===activeCat?'active':''}" onclick="switchCat('${c.id}')">${c.name}</div>`).join('');

}

function switchCat(id){ activeCat=id; renderCatBar(); renderDishes(); window.scrollTo(0,0); }

    html+='<div class="dish-list">';

    list.forEach(d=>{

        const qty=cart[d.name]?cart[d.name].qty:0;

        html+=`

        <div class="dish-card">

function toggleDetail(btn){

    const c=btn.nextElementSibling;

    c.classList.toggle('show');

    btn.textContent=c.classList.contains('show')?'收起 ▴':'详情 ▾';

}

 

function changeQty(name,delta){

    const d=DISHES.find(x=>x.name===name);

    if(!d) return;

    if(cart[name]){

        cart[name].qty+=delta;

        if(cart[name].qty<=0) delete cart[name];

    } else if(delta>0){

        cart[name]={name,price:d.price,unit:d.unit||'',qty:1};

    }

    renderDishes();

    updateCartBar();

}

function updateCartBar(){

    const items=Object.values(cart);

    const count=items.reduce((s,x)=>s+x.qty,0);

    const total=items.reduce((s,x)=>s+x.qty*x.price,0);

    const bar=document.getElementById('cartBar');

    if(count>0){

        bar.classList.remove('hidden');

        document.getElementById('cartCount').textContent=count+'件';

        document.getElementById('cartTotal').textContent='¥'+total;

    } else bar.classList.add('hidden');

}

function toggleCart(show){

    document.getElementById('overlay').classList.toggle('show',show);

    document.getElementById('cartDrawer').classList.toggle('show',show);

    if(show) renderCart();

}

function renderCart(){

    const items=Object.values(cart);

    const box=document.getElementById('cartItems');

    box.innerHTML = items.length

        ? items.map(it=>`<div class="cart-item">

            <div><div class="cart-item-name">${it.name}</div><div class="cart-item-price">¥${it.price}${it.unit||''} × ${it.qty}</div></div>

            <div class="qty-box">

                <button class="qty-btn" onclick="changeQty('${it.name}',-1)">−</button>

                <span class="qty-num">${it.qty}</span>

                <button class="qty-btn" onclick="changeQty('${it.name}',1)">＋</button>

            </div></div>`).join('')

        : '<div class="empty-cart">购物车还是空的</div>';

    const total=items.reduce((s,x)=>s+x.qty*x.price,0);

    document.getElementById('cartTotal2').textContent='¥'+total;

}

 

function showCheckout(){

    const items=Object.values(cart);

    if(!items.length) return;

    toggleCart(false);

    const total=items.reduce((s,x)=>s+x.qty*x.price,0);

    document.getElementById('checkoutTitle').textContent = mode==='dinein'?'堂食订单确认':'客房点餐确认';

    document.getElementById('checkoutDetail').innerHTML =

        items.map(it=>`${it.name} × ${it.qty}　¥${it.price*it.qty}${it.unit||''}`).join('<br>')

        +`<br><br><strong>合计：¥${total}</strong>`;

    document.getElementById('checkoutExtra').innerHTML = mode==='dinein'

        ? '<label>桌号</label><input type="text" id="orderTable" placeholder="如：A5桌">'

        : '<label>房号</label><input type="text" id="orderTable" placeholder="如：桂影阁301">';

    document.getElementById('checkoutModal').classList.add('show');

}

function submitOrder(){

    const items=Object.values(cart);

    const total=items.reduce((s,x)=>s+x.qty*x.price,0);

    const table=document.getElementById('orderTable')?document.getElementById('orderTable').value:'';

    document.getElementById('checkoutModal').classList.remove('show');

    const orderNo='WS'+Date.now().toString().slice(-6);

    document.getElementById('successTitle').textContent='✓ 下单成功';

    document.getElementById('successMsg').innerHTML =

        `订单号 <strong>${orderNo}</strong><br>${mode==='dinein'?'堂食':'客房点餐'} · ${table||'未填'}<br>共 ${items.reduce((s,x)=>s+x.qty,0)} 件 · 合计 <strong>¥${total}</strong><br><br>小二收到后即刻准备，请稍候。`;

    document.getElementById('successModal').classList.add('show');

    cart={};

    renderDishes();

    updateCartBar();

    document.getElementById('orderNote').value='';

}

function closeSuccess(){ document.getElementById('successModal').classList.remove('show'); goCover(); }
